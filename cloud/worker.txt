  async fetch(req, env, ctx) {
    const url = new URL(req.url);
    try {
      if (req.method === 'OPTIONS') return new Response(null, { headers: CORS });
      if (url.pathname === '/api/sync' && req.method === 'POST') return await syncApi(req, env);
      if (url.pathname.startsWith('/api/admin/') && req.method === 'POST') return await adminApi(url.pathname, req, env);
      if ((url.pathname === '/api/info' || url.pathname === '/api/join' || url.pathname === '/api/feedback') && req.method === 'POST') return await publicApi(url.pathname, req, env);
      if (url.pathname.startsWith('/api/') && req.method === 'POST') return await cloudApi(url.pathname, req, env);
      if (url.pathname === '/api/stats' && req.method === 'GET') { await initDb(env); // public totals for the website counter (numbers only, no names)
        const n = async q => (await env.DB.prepare(q).first()).n;
        return new Response(JSON.stringify({ ok: true, members: await n('SELECT COUNT(*) n FROM profiles WHERE banned=0'), friendships: await n("SELECT COUNT(*) n FROM friends WHERE status='accepted'"), follows: await n('SELECT COUNT(*) n FROM follows') }), { headers: Object.assign({ 'content-type': 'application/json', 'cache-control': 'public, max-age=60' }, CORS) }); }
      if (url.pathname === '/admin') return new Response(ADMIN_HTML, { headers: { 'content-type': 'text/html; charset=utf-8' } });
      if (req.method === 'POST') return await interaction(req, env, ctx);
      if (url.pathname === '/setup') return await setupPage(url, env);
      if (url.pathname === '/register') return await register(url, env);
      return new Response(`Nova Cloud v${VERSION} is online. ✨`, { headers: { 'content-type': 'text/plain; charset=utf-8' } });
    } catch (e) {
      console.log('fetch error', e && e.stack || e);
      return new Response('error', { status: 500 });
    }
  },
  async scheduled(event, env, ctx) {
    ctx.waitUntil(fireReminders(env));
  },
};

function hexToBytes(h) { const a = new Uint8Array(h.length / 2); for (let i = 0; i < a.length; i++) a[i] = parseInt(h.substr(i * 2, 2), 16); return a; }
async function verify(req, env, body) {
  const sig = req.headers.get('x-signature-ed25519'), ts = req.headers.get('x-signature-timestamp');
  if (!sig || !ts || !env.DISCORD_PUBLIC_KEY) return false;
  try {
    const key = await crypto.subtle.importKey('raw', hexToBytes(env.DISCORD_PUBLIC_KEY.trim()), { name: 'Ed25519' }, false, ['verify']);
    return await crypto.subtle.verify({ name: 'Ed25519' }, key, hexToBytes(sig), new TextEncoder().encode(ts + body));
  } catch (e) {
    try { // older runtimes
      const key = await crypto.subtle.importKey('raw', hexToBytes(env.DISCORD_PUBLIC_KEY.trim()), { name: 'NODE-ED25519', namedCurve: 'NODE-ED25519' }, false, ['verify']);
      return await crypto.subtle.verify('NODE-ED25519', key, hexToBytes(sig), new TextEncoder().encode(ts + body));
    } catch (e2) { return false; }
  }
}

// ---------------------------------------------------------------- time zones
function parts(tz, d = new Date()) {
  const p = Object.fromEntries(new Intl.DateTimeFormat('en-US', { timeZone: tz, hourCycle: 'h23', year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', weekday: 'long' })
    .formatToParts(d).map(x => [x.type, x.value]));
  return { y: +p.year, m: +p.month, d: +p.day, h: +p.hour % 24, mi: +p.minute, weekday: p.weekday };
}
function toUtc(tz, y, m, d, h, mi) {
  let t = Date.UTC(y, m - 1, d, h, mi);
  for (let i = 0; i < 3; i++) { const p = parts(tz, new Date(t)); const diff = Date.UTC(p.y, p.m - 1, p.d, p.h, p.mi) - Date.UTC(y, m - 1, d, h, mi); if (!diff) break; t -= diff; }
  return t;
}
const pad = n => String(n).padStart(2, '0');
function validTz(tz) { try { new Intl.DateTimeFormat('en-US', { timeZone: tz }); return true; } catch (e) { return false; } }
function niceTime(t, tz) { return new Date(t).toLocaleString('en-US', { timeZone: tz, weekday: 'short', month: 'short', day: 'numeric', hour: 'numeric', minute: '2-digit' }); }
function parseClock(s) {
  s = String(s || '').trim().toLowerCase().replace(/^(\d{1,2})(\d{2})\s*([ap])/, '$1:$2$3').replace(/\s+/g, '');
  if (s === 'noon') return [12, 0]; if (s === 'midnight') return [0, 0];
  const m = s.match(/^(\d{1,2})(?::(\d{2}))?(a\.?m\.?|p\.?m\.?)?$/); if (!m) return null;
  let h = +m[1], mi = +(m[2] || 0); const ap = m[3] || '';
  if (ap.startsWith('p') && h < 12) h += 12; if (ap.startsWith('a') && h === 12) h = 0;
  return h > 23 || mi > 59 ? null : [h, mi];
}
// "6pm", "18:30", "in 20 minutes", "in 2 hours", "tomorrow 9am", "2026-11-07 14:00", "friday 5pm"
function parseWhen(when, tz, now = Date.now()) {
  let s = String(when || '').toLowerCase().trim().replace(/[‘’]/g, "'").replace(/^at\s+/, '');
  const rel = s.match(/^in\s+(\d+(?:\.\d+)?)\s*(m|min|mins|minute|minutes|h|hr|hrs|hour|hours|d|day|days)$/);
  if (rel) { const n = +rel[1], u = rel[2][0]; return now + n * (u === 'm' ? 60000 : u === 'h' ? 3600000 : 86400000); }
  const p = parts(tz, new Date(now));
  let y = p.y, mo = p.m, d = p.d, explicitDay = false;
  const days = ['sunday', 'monday', 'tuesday', 'wednesday', 'thursday', 'friday', 'saturday'];
  let m;
  if ((m = s.match(/^(today|tonight|tomorrow)\s*(?:at\s*)?(.*)$/))) { if (m[1] === 'tomorrow') { const t = new Date(Date.UTC(y, mo - 1, d + 1)); y = t.getUTCFullYear(); mo = t.getUTCMonth() + 1; d = t.getUTCDate(); } explicitDay = true; s = m[2] || (m[1] === 'tonight' ? '8pm' : '9am'); }
  else if ((m = s.match(/^(?:on\s+|next\s+)?(sunday|monday|tuesday|wednesday|thursday|friday|saturday)\s*(?:at\s*)?(.*)$/))) {
    const target = days.indexOf(m[1]), cur = days.indexOf(p.weekday.toLowerCase()); let add = (target - cur + 7) % 7; if (add === 0) add = 7;
    const t = new Date(Date.UTC(y, mo - 1, d + add)); y = t.getUTCFullYear(); mo = t.getUTCMonth() + 1; d = t.getUTCDate(); explicitDay = true; s = m[2] || '9am';
  }
  else if ((m = s.match(/^(\d{4})-(\d{2})-(\d{2})\s*(?:at\s*)?(.*)$/))) { y = +m[1]; mo = +m[2]; d = +m[3]; explicitDay = true; s = m[4] || '9am'; }
  const c = parseClock(s); if (!c) return null;
  let t = toUtc(tz, y, mo, d, c[0], c[1]);
  if (!explicitDay && t <= now) { const n = new Date(Date.UTC(y, mo - 1, d + 1)); t = toUtc(tz, n.getUTCFullYear(), n.getUTCMonth() + 1, n.getUTCDate(), c[0], c[1]); }
  return t;
}
function nextDaily(time, tz, after) {
  const c = parseClock(time); const p = parts(tz, new Date(after));
  for (let add = 0; add < 3; add++) { const n = new Date(Date.UTC(p.y, p.m - 1, p.d + add)); const t = toUtc(tz, n.getUTCFullYear(), n.getUTCMonth() + 1, n.getUTCDate(), c[0], c[1]); if (t > after + 30000) return t; }
  return after + 86400000;
}

// ---------------------------------------------------------------- reminders
async function addReminder(env, u, text, when, daily) {
  const due = parseWhen(when, u.tz);
  if (!due) throw new Error(`couldn't understand the time "${when}" (try 6pm, 18:30, in 20 minutes, tomorrow 9am)`);
  if (due - Date.now() > 366 * 86400000) throw new Error('that is more than a year away');
  const count = await env.DB.prepare('SELECT COUNT(*) AS n FROM reminders WHERE user=? AND (sent IS NULL OR sent=0)').bind(u.id).first();
  if (count && count.n >= 100) throw new Error('you already have 50 reminders, cancel some first');
  const c = parseClock(String(when).replace(/^.*?(\d{1,2}(:\d{2})?\s*(am|pm)?)$/i, '$1'));
  const time = daily ? (c ? pad(c[0]) + ':' + pad(c[1]) : (() => { const p = parts(u.tz, new Date(due)); return pad(p.h) + ':' + pad(p.mi); })()) : '';
  const lp = parts(u.tz, new Date(due));
  const r = await env.DB.prepare('INSERT INTO reminders (user,text,due,daily,time,tz,created,date,sent) VALUES (?,?,?,?,?,?,?,?,0) RETURNING id')
    .bind(u.id, String(text).slice(0, 200), due, daily ? 1 : 0, time || (pad(lp.h) + ':' + pad(lp.mi)), u.tz, Date.now(), daily ? null : `${lp.y}-${pad(lp.m)}-${pad(lp.d)}`).first();
  await env.DB.prepare('UPDATE reminders SET uid=? WHERE id=?').bind('d' + r.id, r.id).run();
  return { id: r.id, text, when: niceTime(due, u.tz), daily: !!daily };
}
async function moveDailyToTz(env, u) {
  const { results } = await env.DB.prepare("SELECT * FROM reminders WHERE user=? AND daily=1 AND time IS NOT NULL AND time != ''").bind(u.id).all();
  for (const r of results) await env.DB.prepare('UPDATE reminders SET tz=?, due=? WHERE id=?').bind(u.tz, nextDaily(r.time, u.tz, Date.now()), r.id).run();
}
async function listReminders(env, u) {
  const { results } = await env.DB.prepare('SELECT * FROM reminders WHERE user=? AND (sent IS NULL OR sent=0) ORDER BY due IS NULL, due').bind(u.id).all();
  return results.map(r => ({ id: r.id, text: r.text, next: r.due ? niceTime(r.due, u.tz) : 'no time set', daily: !!r.daily }));
}
async function cancelReminder(env, u, id) {
  if (String(id).toLowerCase() === 'all') { const r = await env.DB.prepare('DELETE FROM reminders WHERE user=?').bind(u.id).run(); return { cancelled: r.meta && r.meta.changes || 'all' }; }
  const r = await env.DB.prepare('DELETE FROM reminders WHERE user=? AND id=?').bind(u.id, parseInt(String(id).replace(/\D/g, ''), 10) || -1).run();
  if (!(r.meta && r.meta.changes)) throw new Error('no reminder with that number - check /reminders');
  return { cancelled: id };
}
async function updateReminder(env, u, id, text, when, daily) {
  const r = await env.DB.prepare('SELECT * FROM reminders WHERE user=? AND id=?').bind(u.id, parseInt(String(id).replace(/\D/g, ''), 10) || -1).first();
  if (!r) throw new Error('no reminder with that number - call list_reminders first');
  let due = r.due, time = r.time, dly = daily === undefined || daily === null ? r.daily : (daily ? 1 : 0);
  if (when) { const t = parseWhen(when, u.tz); if (!t) throw new Error(`couldn't understand the time "${when}"`); due = t; const p = parts(u.tz, new Date(t)); time = pad(p.h) + ':' + pad(p.mi); }
  if (dly && !time) { const p = parts(u.tz, new Date(due)); time = pad(p.h) + ':' + pad(p.mi); }
  const lp = parts(u.tz, new Date(due));
  await env.DB.prepare('UPDATE reminders SET text=?, due=?, daily=?, time=?, date=?, sent=0 WHERE id=?').bind(text ? String(text).slice(0, 200) : r.text, due, dly, time, dly ? null : `${lp.y}-${pad(lp.m)}-${pad(lp.d)}`, r.id).run();
  return { updated: r.id, text: text || r.text, when: niceTime(due, u.tz), daily: !!dly };
}
const DING = ['⏰ **Reminder:** {t}', '⏰ {t}. You asked me to nag you, so: consider yourself nagged.', '⏰ **{t}**. This is your friendly neighbourhood AI, right on time as usual.',
  '⏰ Hey. **{t}**. I\'d do it for you, but I don\'t have hands.', '⏰ Reminder: **{t}**. Future you says thanks. Probably.'];
async function fireReminders(env) {
  await initDb(env);
  const now = Date.now();
  const { results } = await env.DB.prepare('SELECT * FROM reminders WHERE due IS NOT NULL AND due <= ? AND (sent IS NULL OR sent=0) ORDER BY due LIMIT 40').bind(now).all();
  for (const r of results) {
    // claim it first, so it can never be sent twice
    const claim = r.daily ? await env.DB.prepare('UPDATE reminders SET due=? WHERE id=? AND due=?').bind(nextDaily(r.time, r.tz, now), r.id, r.due).run()
                          : await env.DB.prepare('UPDATE reminders SET sent=1 WHERE id=? AND due=? AND (sent IS NULL OR sent=0)').bind(r.id, r.due).run();
    if (!(claim.meta && claim.meta.changes)) continue;
    const late = now - r.due > 3 * 3600000; // way overdue (e.g. service was down): skip one-offs quietly
    try { if (!late || r.daily) await deliver(env, r, r.uid && r.uid[0] === 'x' ? '⏰ ' + r.text : DING[Math.floor(Math.random() * DING.length)].replace('{t}', r.text), !!r.daily); }
    catch (e) { console.log('deliver failed', r.user, e.message); }
  }
  // tidy up: forget fired one-time reminders after 60 days
  if (Math.random() < 0.02) await env.DB.prepare('DELETE FROM reminders WHERE sent=1 AND due < ?').bind(now - 60 * 86400000).run();
}

async function groq(env, messages, tools) {
  const r = await fetch((env.GROQ_URL || 'https://api.groq.com/openai/v1') + '/chat/completions', { method: 'POST',
    headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + env.GROQ_KEY },
    body: JSON.stringify(Object.assign({ model: env.GROQ_MODEL || 'openai/gpt-oss-120b', messages, temperature: 0.9, max_tokens: 700 }, tools ? { tools, tool_choice: 'auto' } : {})) });
  const j = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error('groq ' + r.status + ' ' + (j.error && j.error.message || ''));
  return j.choices[0].message;
}
async function gemini(env, body) {
  const model = env.GEMINI_MODEL || 'gemini-flash-latest';
  const r = await fetch(`${env.GEMINI_URL || 'https://generativelanguage.googleapis.com/v1beta'}/models/${model}:generateContent`, { method: 'POST',
    headers: { 'Content-Type': 'application/json', 'x-goog-api-key': env.GEMINI_KEY }, body: JSON.stringify(body) });
  const j = await r.json().catch(() => ({}));
  if (!r.ok) throw new Error('gemini ' + r.status + ' ' + (j.error && j.error.message || ''));
  return j;
}
async function askAI(env, prompt) {
  if (env.GROQ_KEY) { try { const m = await groq(env, [{ role: 'user', content: prompt }], undefined); if (m && m.content) return String(m.content).trim(); } catch (e) { console.log('askAI groq', e.message); } }
  if (env.GEMINI_KEY) { try { const j = await gemini(env, { contents: [{ role: 'user', parts: [{ text: prompt }] }], generationConfig: { temperature: 1 } }); return ((j.candidates || [])[0]?.content?.parts || []).map(p => p.text || '').join('').trim(); } catch (e) { console.log('askAI gemini', e.message); } }
  return '';
}
const STYLE = 'You are NOVA, a witty, sarcastic-but-kind AI in a Discord server. Keep it short, clean and clever. Plain Discord text, no headers.';
async function think(env, u, text, extra) {
  const T = makeTools(env, u);
  const p = parts(u.tz);
  const sys = PERSONA(u, `${p.weekday}, ${niceTime(Date.now(), u.tz)}`, u.tz, u.facts.map(f => '- ' + f).join('\n') || '- (nothing yet)', extra);
  const hist = u.history.slice(-12);
  const actions = [];
  const run = async (name, args) => { let res; try { if (!T[name]) throw new Error('unknown tool'); res = await T[name].run(args || {}); } catch (e) { res = { error: e.message || String(e) }; } actions.push(name); return res; };
  const brains = [env.GROQ_KEY && 'groq', env.GEMINI_KEY && 'gemini'].filter(Boolean);
  if (!brains.length) return 'I have no brain configured. Someone needs to add a GROQ_KEY in Cloudflare. Rude, honestly.';
  for (const brain of brains) {
    try {
      if (brain === 'groq') {
        const msgs = [{ role: 'system', content: sys }, ...hist.map(h => ({ role: h.r === 'u' ? 'user' : 'assistant', content: h.t })), { role: 'user', content: text }];
        for (let i = 0; i < 6; i++) {
          const m = await groq(env, msgs, oaTools(T));
          if (!m.tool_calls || !m.tool_calls.length) return String(m.content || '').trim() || 'Done.';
          msgs.push({ role: 'assistant', content: m.content || null, tool_calls: m.tool_calls });
          for (const c of m.tool_calls) { let a = {}; try { a = JSON.parse(c.function.arguments || '{}'); } catch (e) {} msgs.push({ role: 'tool', tool_call_id: c.id, content: JSON.stringify(await run(c.function.name, a)).slice(0, 6000) }); }
        }
      } else {
        const contents = [...hist.map(h => ({ role: h.r === 'u' ? 'user' : 'model', parts: [{ text: h.t }] })), { role: 'user', parts: [{ text }] }];
        while (contents.length && contents[0].role !== 'user') contents.shift();
        const decl = [{ functionDeclarations: Object.entries(T).map(([name, t]) => ({ name, description: t.d, ...(Object.keys(t.p).length ? { parameters: { type: 'OBJECT', properties: Object.fromEntries(Object.entries(t.p).map(([k, v]) => [k, { type: v.toUpperCase() }])), required: t.r || [] } } : {}) })) }];
        for (let i = 0; i < 6; i++) {
          const j = await gemini(env, { systemInstruction: { parts: [{ text: sys }] }, contents, tools: decl, generationConfig: { temperature: 0.9 } });
          const c = j.candidates && j.candidates[0]; const ps = (c && c.content && c.content.parts) || [];
          const calls = ps.filter(x => x.functionCall);
          if (!calls.length) return ps.map(x => x.text || '').join('').trim() || 'Done.';
          contents.push(c.content);
          const resp = []; for (const x of calls) resp.push({ functionResponse: { name: x.functionCall.name, response: { result: await run(x.functionCall.name, x.functionCall.args) } } });
          contents.push({ role: 'user', parts: resp });
        }
      }
      return 'I got a bit tangled on that one. Try saying it another way?';
    } catch (e) {
      console.log('brain failed', brain, e.message);
      if (actions.length) return 'Done, but my brain cut out before I could say anything clever about it.';
    }
  }
  return 'My brain is taking a nap (the AI service is down or out of free uses for today). Try again in a bit. Reminders still go out on time.';
}

// ---------------------------------------------------------------- handling interactions
const opt = (data, name) => { const o = (data.options || []).find(x => x.name === name); return o ? o.value : undefined; };
async function interaction(req, env, ctx) {
  const body = await req.text();
  if (!(await verify(req, env, body))) return new Response('bad signature', { status: 401 });
  const i = JSON.parse(body);
  if (i.type === 1) return json({ type: 1 }); // Discord's PING
  if (i.type !== 2) return json({ type: 4, data: { content: 'Hm?' } });
  const who = i.member ? i.member.user : i.user;
  const name = (i.member && i.member.nick) || who.global_name || who.username;
  const cmd = i.data.name;
  const priv = ['reminders', 'cancel', 'timezone', 'reset', 'link'].includes(cmd);
  i.__origin = new URL(req.url).origin;
  ctx.waitUntil(handle(env, i, who.id, name, cmd).then(out => followup(env, i.token, out)).catch(async e => {
    console.log('handle error', e && e.stack || e);
    try { await followup(env, i.token, 'Something on my end broke. Not my proudest moment. Try again?'); } catch (e2) {}
  }));
  return json({ type: 5, data: priv ? { flags: 64 } : {} }); // "Nova is thinking…"
}

// ---------------------------------------------------------------- phone <-> Discord sync
async function syncApi(req, env) {
  let b; try { b = await req.json(); } catch (e) { return new Response(JSON.stringify({ error: 'bad request' }), { status: 400, headers: H }); }
  await initDb(env);
  const code = String(b.code || '').trim().toUpperCase();
  const row = code.length >= 10 ? await env.DB.prepare('SELECT id FROM users WHERE link=?').bind(code).first() : null;
  if (!row) return new Response(JSON.stringify({ error: 'unknown link code. Run /link in Discord to get yours.' }), { status: 403, headers: H });
  const u = await getUser(env, row.id);
  if (b.tz && validTz(b.tz) && b.tz !== u.tz) { u.tz = b.tz; await moveDailyToTz(env, u); }
  if (b.name && !u.name) u.name = String(b.name).slice(0, 40);
  const now = Date.now();
  // apply phone changes
  for (const it of (Array.isArray(b.changed) ? b.changed.slice(0, 300) : [])) {
    const uid = String(it.uid || '').slice(0, 40); if (!uid) continue;
    const text = String(it.text || '').slice(0, 200) || 'Untitled';
    const time = parseClock(it.time) ? (() => { const c = parseClock(it.time); return pad(c[0]) + ':' + pad(c[1]); })() : '';
    const daily = it.kind === 'task' ? 1 : 0;
    const date = daily ? null : (/^\d{4}-\d{2}-\d{2}$/.test(it.date || '') ? it.date : null);
    let due = null, sent = 0;
    if (daily && time) due = nextDaily(time, u.tz, now - 60000);
    if (!daily && date && time) { const [y, m, dd] = date.split('-').map(Number); const c = parseClock(time); due = toUtc(u.tz, y, m, dd, c[0], c[1]); if (due <= now) sent = 1; }
    if (!daily && date && !time) sent = 0;
    const ex = await env.DB.prepare('SELECT id FROM reminders WHERE user=? AND uid=?').bind(u.id, uid).first();
    if (ex) await env.DB.prepare('UPDATE reminders SET text=?, due=?, daily=?, time=?, tz=?, date=?, sent=? WHERE id=?').bind(text, due, daily, time, u.tz, date, sent, ex.id).run();
    else await env.DB.prepare('INSERT INTO reminders (user,text,due,daily,time,tz,created,date,sent,uid) VALUES (?,?,?,?,?,?,?,?,?,?)').bind(u.id, text, due, daily, time, u.tz, now, date, sent, uid).run();
  }
  for (const uid of (Array.isArray(b.deleted) ? b.deleted.slice(0, 300) : [])) await env.DB.prepare('DELETE FROM reminders WHERE user=? AND uid=?').bind(u.id, String(uid)).run();
  // facts
  const low = f => String(f).toLowerCase();
  for (const f of (Array.isArray(b.factsAdd) ? b.factsAdd.slice(0, 100) : [])) { const t = String(f).trim().slice(0, 200); if (t && !u.facts.some(x => low(x) === low(t))) u.facts.push(t); }
  if (Array.isArray(b.factsDel)) { const del = new Set(b.factsDel.map(low)); u.facts = u.facts.filter(x => !del.has(low(x))); }
  await saveUser(env, u);
  // send back everything for this user
  const { results } = await env.DB.prepare('SELECT * FROM reminders WHERE user=? AND (sent IS NULL OR sent=0 OR due > ?) ORDER BY id').bind(u.id, now - 45 * 86400000).all();
  const items = [];
  for (const r of results) {
    if (r.uid && r.uid[0] === 'x') continue; // timer alerts stay out of the schedule
    if (!r.uid) { r.uid = 'd' + r.id; await env.DB.prepare('UPDATE reminders SET uid=? WHERE id=?').bind(r.uid, r.id).run(); }
    let date = r.date, time = r.time || '';
    if (!r.daily && !date && r.due) { const p = parts(u.tz, new Date(r.due)); date = `${p.y}-${pad(p.m)}-${pad(p.d)}`; time = time || pad(p.h) + ':' + pad(p.mi); }
    items.push(r.daily ? { uid: r.uid, kind: 'task', text: r.text, time } : { uid: r.uid, kind: 'event', text: r.text, date: date || '', time });
  }
  return new Response(JSON.stringify({ ok: true, name: u.name || '', tz: u.tz, items, facts: u.facts, features: features(env), vapid: (await vapid(env)).pub }), { headers: H });
}

// ---- Web Push (RFC 8291 + VAPID), all with built-in WebCrypto
const b64u = b => btoa(String.fromCharCode(...new Uint8Array(b))).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
const unb64u = s => { s = String(s).replace(/-/g, '+').replace(/_/g, '/'); while (s.length % 4) s += '='; return Uint8Array.from(atob(s), c => c.charCodeAt(0)); };
const cat = (...a) => { const o = new Uint8Array(a.reduce((n, x) => n + x.length, 0)); let i = 0; for (const x of a) { o.set(x, i); i += x.length; } return o; };
const utf8 = s => new TextEncoder().encode(s);
async function hkdf(salt, ikm, info, len) {
  const k = await crypto.subtle.importKey('raw', ikm, 'HKDF', false, ['deriveBits']);
  return new Uint8Array(await crypto.subtle.deriveBits({ name: 'HKDF', hash: 'SHA-256', salt, info }, k, len * 8));
}
async function vapid(env) {
  let v = await kvGet(env, 'vapid');
  if (!v) {
    const kp = await crypto.subtle.generateKey({ name: 'ECDSA', namedCurve: 'P-256' }, true, ['sign', 'verify']);
    v = { pub: b64u(await crypto.subtle.exportKey('raw', kp.publicKey)), jwk: await crypto.subtle.exportKey('jwk', kp.privateKey) };
    await kvSet(env, 'vapid', v);
  }
  return v;
}
async function encryptPush(sub, payload) {
  const ua = unb64u(sub.p256dh), auth = unb64u(sub.auth);
  const eph = await crypto.subtle.generateKey({ name: 'ECDH', namedCurve: 'P-256' }, true, ['deriveBits']);
  const asPub = new Uint8Array(await crypto.subtle.exportKey('raw', eph.publicKey));
  const uaKey = await crypto.subtle.importKey('raw', ua, { name: 'ECDH', namedCurve: 'P-256' }, false, []);
  const secret = new Uint8Array(await crypto.subtle.deriveBits({ name: 'ECDH', public: uaKey }, eph.privateKey, 256));
  const ikm = await hkdf(auth, secret, cat(utf8('WebPush: info\0'), ua, asPub), 32);
  const salt = crypto.getRandomValues(new Uint8Array(16));
  const cek = await hkdf(salt, ikm, utf8('Content-Encoding: aes128gcm\0'), 16);
  const nonce = await hkdf(salt, ikm, utf8('Content-Encoding: nonce\0'), 12);
  const key = await crypto.subtle.importKey('raw', cek, 'AES-GCM', false, ['encrypt']);
  const ct = new Uint8Array(await crypto.subtle.encrypt({ name: 'AES-GCM', iv: nonce }, key, cat(utf8(payload), new Uint8Array([2]))));
  return cat(salt, new Uint8Array([0, 0, 16, 0]), new Uint8Array([asPub.length]), asPub, ct);
}
async function vapidAuth(env, endpoint) {
  const v = await vapid(env);
  const key = await crypto.subtle.importKey('jwk', v.jwk, { name: 'ECDSA', namedCurve: 'P-256' }, false, ['sign']);
  const head = b64u(utf8(JSON.stringify({ typ: 'JWT', alg: 'ES256' })));
  const body = b64u(utf8(JSON.stringify({ aud: new URL(endpoint).origin, exp: Math.floor(Date.now() / 1000) + 12 * 3600, sub: 'mailto:' + (env.GMAIL_USER || 'nova@example.com') })));
  const sig = await crypto.subtle.sign({ name: 'ECDSA', hash: 'SHA-256' }, key, utf8(head + '.' + body));
  return `vapid t=${head}.${body}.${b64u(sig)}, k=${v.pub}`;
}
async function sendPush(env, sub, msg) {
  const r = await fetch(sub.endpoint, { method: 'POST', body: await encryptPush(sub, JSON.stringify(msg)),
    headers: { 'Content-Encoding': 'aes128gcm', 'Content-Type': 'application/octet-stream', TTL: '86400', Urgency: 'high', Authorization: await vapidAuth(env, sub.endpoint) } });
  if (r.status === 404 || r.status === 410) await env.DB.prepare('DELETE FROM subs WHERE endpoint=?').bind(sub.endpoint).run(); // phone unsubscribed
  return r.ok;
}
async function pushUser(env, userId, msg) {
  const { results } = await env.DB.prepare('SELECT * FROM subs WHERE user=?').bind(userId).all();
  let ok = 0; for (const s of results) { try { if (await sendPush(env, s, msg)) ok++; } catch (e) { console.log('push failed', e.message); } }
  return ok;
}

const b64utf8 = s => { const b = utf8(s); let o = ''; for (let i = 0; i < b.length; i += 0x8000) o += String.fromCharCode(...b.subarray(i, i + 0x8000)); return btoa(o); };
async function smtpSend(env, to, subject, text, fromName) {
  const user = String(env.GMAIL_USER).trim(), pass = String(env.GMAIL_APP_PASSWORD).replace(/\s+/g, '');
  const sock = connect({ hostname: env.SMTP_HOST || 'smtp.gmail.com', port: +(env.SMTP_PORT || 465) }, { secureTransport: 'on' });
  const w = sock.writable.getWriter(), rd = sock.readable.getReader(), dec = new TextDecoder();
  let buf = '';
  const read = async () => {
    for (;;) {
      const lines = buf.split('\r\n');
      for (let i = 0; i < lines.length - 1; i++) if (/^\d{3} /.test(lines[i])) { buf = lines.slice(i + 1).join('\r\n'); return lines.slice(0, i + 1).join('\n'); }
      const { value, done } = await rd.read(); if (done) throw new Error('mail server hung up'); buf += dec.decode(value, { stream: true });
    }
  };
  const step = async (line, ok) => { if (line !== null) await w.write(utf8(line + '\r\n')); const rep = await read(); if (!rep.startsWith(String(ok))) throw new Error('Gmail said: ' + rep.replace(/\s+/g, ' ').slice(0, 180)); return rep; };
  try {
    await step(null, 220); await step('EHLO nova.cloud', 250); await step('AUTH LOGIN', 334); await step(btoa(user), 334);
    try { await step(btoa(pass), 235); } catch (e) { throw new Error('Gmail refused the login. Check GMAIL_USER and GMAIL_APP_PASSWORD (an App Password, not your normal password).'); }
    await step(`MAIL FROM:<${user}>`, 250);
    for (const t of to) await step(`RCPT TO:<${t}>`, 250);
    await step('DATA', 354);
    const body = b64utf8(text).replace(/.{76}/g, '$&\r\n');
    const msg = [`From: =?UTF-8?B?${b64utf8(fromName || 'Nova')}?= <${user}>`, `To: ${to.join(', ')}`, `Subject: =?UTF-8?B?${b64utf8(subject)}?=`, `Date: ${new Date().toUTCString()}`,
      `Message-ID: <${crypto.randomUUID()}@nova.cloud>`, 'MIME-Version: 1.0', 'Content-Type: text/plain; charset=UTF-8', 'Content-Transfer-Encoding: base64', '', body].join('\r\n');
    await step(msg + '\r\n.', 250);
    try { await step('QUIT', 221); } catch (e) {}
  } finally { try { rd.releaseLock(); w.releaseLock(); await sock.close(); } catch (e) {} }
}

    const day = 'mail:' + new Date().toISOString().slice(0, 10), n = (await kvGet(env, day)) || 0;

// can this viewer see this post? (yours, a friend's, or someone you follow; never blocked; 21+ only for 21+)
async function canSeePost(env, post, me, meProf) {
  if (!post || post.removed) return false;
  if (post.author === me) return true;
  if (post.adult && ageOf(meProf && meProf.dob) < 21) return false;
  if (await blockedEither(env, me, post.author)) return false;
  if (await areFriends(env, me, post.author)) return true;
  if (await env.DB.prepare('SELECT 1 FROM follows WHERE a=? AND b=?').bind(me, post.author).first()) return true;
  if (post.repost_of) return false;
  // someone you're connected to reposted it
  const { results } = await env.DB.prepare('SELECT * FROM posts WHERE repost_of=? AND removed=0').bind(post.id).all();
  for (const r of results) if (await canSeePost(env, r, me, meProf)) return true;
  return false;
}

}
async function canSeePhoto(env, ph, me, meProf) {
  if (!ph) return false;
  if (ph.owner === me) return true;
  if (ph.status !== 'ok') return false;
  if (ph.adult && ageOf(meProf && meProf.dob) < 21) return false;
  if (await blockedEither(env, me, ph.owner)) return false;
  if (await areFriends(env, me, ph.owner)) return true;
  // profile pictures + banners are visible to people who look you up by exact username / code (so they can add you)
  const asAvatar = await env.DB.prepare('SELECT 1 FROM profiles WHERE (photo=? OR banner=?) AND id=?').bind(ph.id, ph.id, ph.owner).first();
  if (asAvatar) return true;
  if (await env.DB.prepare('SELECT 1 FROM follows WHERE a=? AND b=?').bind(me, ph.owner).first()) return true;
  // a picture in a post a friend reposted
  const { results } = await env.DB.prepare('SELECT * FROM posts WHERE photo=? AND removed=0').bind(ph.id).all();
  for (const post of results) { const { results: rp } = await env.DB.prepare('SELECT * FROM posts WHERE repost_of=? AND removed=0').bind(post.id).all(); for (const r of rp) if (await canSeePost(env, r, me, meProf)) return true; }
  return false;
}
async function makeProfile(env, uid, b, ip) {
  const username = clean(b.username, 20).toLowerCase(), display = clean(b.display, 30) || username, dob = clean(b.dob, 10), age = ageOf(dob);
  if (!okUsername(username)) return { error: 'Usernames are 3–20 characters: letters, numbers, _ or .' };
  if (age < 0 || age > 110) return { error: 'Enter your real birthday.' };
  if (age < 18) return { error: 'Connect is for people 18 and older.', underage: true };
  if (await env.DB.prepare('SELECT 1 FROM profiles WHERE username=?').bind(username).first()) return { error: 'That username is taken.' };
  let fcode; do { fcode = rnd(6); } while (await env.DB.prepare('SELECT 1 FROM profiles WHERE fcode=?').bind(fcode).first());
  await env.DB.prepare('INSERT INTO profiles (id,username,display,bio,dob,photo,fcode,created,banned,ip) VALUES (?,?,?,?,?,?,?,?,0,?)').bind(uid, username, display, clean(b.bio, 160), dob, null, fcode, now(), ip || '').run();
  return { ok: true };
}

// ---- shared brain: friends without their own AI key use Moon's, with a daily limit
const BRAIN_MODELS = ['openai/gpt-oss-120b', 'openai/gpt-oss-20b', 'groq/compound-mini', 'groq/compound', 'llama-3.3-70b-versatile', 'llama-3.1-8b-instant'];
async function brainProxy(env, u, b) {
  if (!env.GROQ_KEY) return jr({ error: { message: 'The shared brain isn\'t set up on Nova Cloud.' } }, 503);
  const p = await getProf(env, u.id); if (p && p.banned) return jr({ error: { message: 'This account is suspended.' } }, 403);
  if (u.id !== 'owner') {
    const lim = +(env.BRAIN_DAILY || 150), row = await env.DB.prepare('SELECT n FROM usage WHERE user=? AND day=?').bind(u.id, today()).first();
    if (row && row.n >= lim) return jr({ error: { message: 'Daily limit reached (' + lim + ' messages). Nova resets at midnight UTC.' } }, 429);
    await env.DB.prepare('INSERT INTO usage (user,day,n) VALUES (?,?,1) ON CONFLICT(user,day) DO UPDATE SET n=n+1').bind(u.id, today()).run();
  }
  const body = Object.assign({}, b.body || {}); if (!BRAIN_MODELS.includes(body.model)) body.model = BRAIN_MODELS[0];
  body.max_tokens = Math.min(+body.max_tokens || 2000, 4000); delete body.stream;
  const r = await fetch('https://api.groq.com/openai/v1/chat/completions', { method: 'POST', headers: { 'Authorization': 'Bearer ' + env.GROQ_KEY, 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
  return new Response(await r.text(), { status: r.status, headers: H });
}
// ---- one post as the app sees it (likes, comments, reposts, and the original if it's a repost)
async function postView(env, r, me, meProf, ph) {
  const one = async (sql, ...a) => (await env.DB.prepare(sql).bind(...a).first()).n;
  const v = { id: r.id, author: pub(await getProf(env, r.author), me), text: r.text, photo: r.photo, photoStatus: ph ? ph.status : null, adult: !!r.adult, at: r.created, mine: r.author === me,
    likes: await one('SELECT COUNT(*) n FROM likes WHERE post=?', r.id), liked: !!(await env.DB.prepare('SELECT 1 FROM likes WHERE post=? AND user=?').bind(r.id, me).first()),
    comments: await one('SELECT COUNT(*) n FROM comments WHERE post=? AND removed=0', r.id), reposts: await one('SELECT COUNT(*) n FROM posts WHERE repost_of=? AND removed=0', r.id) };
  if (r.repost_of) {
    const o = await env.DB.prepare('SELECT * FROM posts WHERE id=?').bind(r.repost_of).first();
    const blocked = o && o.author !== me && await blockedEither(env, me, o.author);
    if (!o || o.removed || blocked || (o.adult && ageOf(meProf && meProf.dob) < 21)) v.original = { gone: true };
    else { const oph = o.photo ? await env.DB.prepare('SELECT status FROM photos WHERE id=?').bind(o.photo).first() : null;
      v.original = { id: o.id, author: pub(await getProf(env, o.author), me), text: o.text, photo: oph && oph.status === 'ok' ? o.photo : null, adult: !!o.adult, at: o.created, mine: o.author === me,
        likes: await one('SELECT COUNT(*) n FROM likes WHERE post=?', o.id), liked: !!(await env.DB.prepare('SELECT 1 FROM likes WHERE post=? AND user=?').bind(o.id, me).first()),
        comments: await one('SELECT COUNT(*) n FROM comments WHERE post=? AND removed=0', o.id), reposts: await one('SELECT COUNT(*) n FROM posts WHERE repost_of=? AND removed=0', o.id) }; }
  }
  return v;
}

// ---- Moon's admin page: reports, pictures waiting for review, users, invites
async function adminApi(path, req, env) {
  let b; try { b = await req.json(); } catch (e) { return jr({ error: 'bad request' }, 400); }
  if (!env.SETUP_KEY || b.key !== env.SETUP_KEY) return jr({ error: 'wrong key' }, 403);
  await initDb(env);
  const one = async sql => (await env.DB.prepare(sql).first()).n;
  if (path === '/api/admin/stats') return jr({ ok: true, users: await one('SELECT COUNT(*) n FROM profiles'), banned: await one('SELECT COUNT(*) n FROM profiles WHERE banned=1'), posts: await one('SELECT COUNT(*) n FROM posts WHERE removed=0'), msgs: await one('SELECT COUNT(*) n FROM msgs'), openReports: await one("SELECT COUNT(*) n FROM reports WHERE status='open'"), pending: await one("SELECT COUNT(*) n FROM photos WHERE status='pending'"), brainToday: await one("SELECT COALESCE(SUM(n),0) n FROM usage WHERE day='" + today() + "'"), feedback: await one('SELECT COUNT(*) n FROM feedback WHERE done=0') });
  if (path === '/api/admin/feedback') { const { results } = await env.DB.prepare('SELECT * FROM feedback WHERE done=0 ORDER BY created DESC LIMIT 100').all(); return jr({ ok: true, feedback: results }); }
  if (path === '/api/admin/feedback-done') { await env.DB.prepare('UPDATE feedback SET done=1 WHERE id=?').bind(+b.id).run(); return jr({ ok: true }); }
  if (path === '/api/admin/reports') {
    const { results } = await env.DB.prepare("SELECT r.*, p.username AS rep FROM reports r LEFT JOIN profiles p ON p.id=r.reporter WHERE r.status='open' ORDER BY r.id DESC LIMIT 100").all();
    for (const r of results) {
      if (r.kind === 'user') { const t = await getProf(env, r.target); r.preview = t ? '@' + t.username + ' · ' + (t.display || '') + (t.bio ? ' · ' + t.bio : '') : '(deleted)'; r.photo = t && t.photo; r.owner = r.target; }
      if (r.kind === 'post') { const t = await env.DB.prepare('SELECT * FROM posts WHERE id=?').bind(+r.target).first(); r.preview = t ? t.text : '(deleted)'; r.photo = t && t.photo; r.owner = t && t.author; }
      if (r.kind === 'msg') { const t = await env.DB.prepare('SELECT * FROM msgs WHERE id=?').bind(+r.target).first(); r.preview = t ? t.text : '(deleted)'; r.owner = t && t.a; }
      if (r.kind === 'comment') { const t = await env.DB.prepare('SELECT * FROM comments WHERE id=?').bind(+r.target).first(); r.preview = t ? 'Comment: ' + t.text : '(deleted)'; r.owner = t && t.author; }
      if (r.kind === 'photo') { r.photo = r.target; const t = await env.DB.prepare('SELECT owner FROM photos WHERE id=?').bind(r.target).first(); r.owner = t && t.owner; }
      if (r.photo) { const ph = await env.DB.prepare('SELECT data FROM photos WHERE id=?').bind(r.photo).first(); r.photoData = ph && ph.data; }
      if (r.owner) { const o = await getProf(env, r.owner); r.ownerName = o ? '@' + o.username : r.owner; }
    }
    return jr({ ok: true, reports: results });
  }
  if (path === '/api/admin/resolve') {
    const r = await env.DB.prepare('SELECT * FROM reports WHERE id=?').bind(+b.id).first(); if (!r) return jr({ error: 'not found' }, 404);
    const mark = st => env.DB.prepare('UPDATE reports SET status=? WHERE kind=? AND target=? AND status=\'open\'').bind(st, r.kind, r.target).run();
    if (b.action === 'dismiss') { if (r.kind === 'post') await env.DB.prepare('UPDATE posts SET removed=0 WHERE id=? AND removed=2').bind(+r.target).run(); if (r.kind === 'msg') await env.DB.prepare('UPDATE msgs SET removed=0 WHERE id=? AND removed=2').bind(+r.target).run(); if (r.kind === 'comment') await env.DB.prepare('UPDATE comments SET removed=0 WHERE id=? AND removed=2').bind(+r.target).run(); await mark('dismissed'); }
    if (b.action === 'remove' || b.action === 'ban') {
      if (r.kind === 'post') await env.DB.prepare('UPDATE posts SET removed=3 WHERE id=?').bind(+r.target).run();
      if (r.kind === 'msg') await env.DB.prepare('UPDATE msgs SET removed=3 WHERE id=?').bind(+r.target).run();
      if (r.kind === 'comment') await env.DB.prepare('UPDATE comments SET removed=3 WHERE id=?').bind(+r.target).run();
      if (r.kind === 'photo') await env.DB.prepare("UPDATE photos SET status='rejected', data='' WHERE id=?").bind(r.target).run();
      if (r.kind === 'user') { const t = await getProf(env, r.target); for (const ph of [t && t.photo, t && t.banner]) if (ph) await env.DB.prepare("UPDATE photos SET status='rejected', data='' WHERE id=?").bind(ph).run(); await env.DB.prepare('UPDATE profiles SET photo=NULL, banner=NULL, bio=\'\' WHERE id=?').bind(r.target).run(); }
      if (b.action === 'ban' && b.owner) await env.DB.prepare('UPDATE profiles SET banned=1 WHERE id=?').bind(String(b.owner)).run();
      await mark(b.action === 'ban' ? 'banned' : 'removed');
    }
    return jr({ ok: true });
  }
  if (path === '/api/admin/photos') { const { results } = await env.DB.prepare("SELECT ph.id, ph.data, ph.adult, ph.reason, ph.created, p.username FROM photos ph LEFT JOIN profiles p ON p.id=ph.owner WHERE ph.status='pending' ORDER BY ph.created LIMIT 50").all(); return jr({ ok: true, photos: results }); }
  if (path === '/api/admin/photo') { await env.DB.prepare(b.ok ? "UPDATE photos SET status='ok' WHERE id=?" : "UPDATE photos SET status='rejected', data='' WHERE id=?").bind(String(b.id)).run(); return jr({ ok: true }); }
  if (path === '/api/admin/users') { const { results } = await env.DB.prepare('SELECT id, username, display, dob, created, banned, (SELECT COUNT(*) FROM reports WHERE target=profiles.id) AS reports FROM profiles ORDER BY created DESC LIMIT 300').all(); return jr({ ok: true, users: results.map(u => Object.assign(u, { age: ageOf(u.dob), dob: undefined })) }); }
  if (path === '/api/admin/ban') { await env.DB.prepare('UPDATE profiles SET banned=? WHERE id=?').bind(b.ban ? 1 : 0, String(b.id)).run(); return jr({ ok: true }); }
  if (path === '/api/admin/invites') { const { results } = await env.DB.prepare('SELECT * FROM invites ORDER BY created DESC LIMIT 200').all(); return jr({ ok: true, invites: results }); }
  if (path === '/api/admin/invite-new') {
    const n = Math.max(1, Math.min(50, +b.n || 1)), max = Math.max(1, Math.min(100, +b.max || 1)), out = [];
    for (let i = 0; i < n; i++) { const c = 'JOIN-' + rnd(6); await env.DB.prepare('INSERT INTO invites (code,created,uses,max,note) VALUES (?,?,0,?,?)').bind(c, now(), max, clean(b.note, 60)).run(); out.push(c); }
    return jr({ ok: true, codes: out });
  }
  if (path === '/api/admin/invite-delete') { await env.DB.prepare('DELETE FROM invites WHERE code=?').bind(String(b.code)).run(); return jr({ ok: true }); }
  return jr({ error: 'unknown endpoint' }, 404);
}
